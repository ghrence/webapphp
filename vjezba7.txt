    <form method="post">
        <label for="kolokvij1">Ocjena I. kolokvija:</label>
        <input type="number" name="kolokvij1" id="kolokvij1" min="1" max="5" step="0.01" required>

        <label for="kolokvij2">Ocjena II. kolokvija:</label>
        <input type="number" name="kolokvij2" id="kolokvij2" min="1" max="5" step="0.01" required>

        <button type="submit" name="submit">Izračunaj</button>
    </form>

    <?php
    if (isset($_POST['submit'])) {

        $kolokvij1 = $_POST['kolokvij1'];
        $kolokvij2 = $_POST['kolokvij2'];

        if ($kolokvij1 < 1 || $kolokvij2 < 1) {
            echo "<h3>Konačna ocjena: 1 (negativna)</h3>";
        } else {
            $prosjek = ($kolokvij1 + $kolokvij2) / 2;

            if ($prosjek >= 4.5) {
                $final_grade = 5;
            } elseif ($prosjek >= 3.5) {
                $final_grade = 4;
            } elseif ($prosjek >= 2.5) {
                $final_grade = 3;
            } elseif ($prosjek >= 1.5) {
                $final_grade = 2;
            } else {
                $final_grade = 1;
            }

            echo "<h3>Prosjek: $prosjek</h3>";
            echo "<h3>Konačna ocjena: $final_grade</h3>";
        }
    }
    ?>
